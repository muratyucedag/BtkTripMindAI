 public IActionResult Index()
        {
            return View();
        }

        [HttpPost]
        public IActionResult SendMessage([FromBody] AgentMessageRequestDto request)
        {
            if (request == null || string.IsNullOrWhiteSpace(request.Message))
            {
                return BadRequest(new AgentMessageResponseDto
                {
                    IsSuccess = false,
                    Message = "Lütfen geçerli bir mesaj giriniz."
                });
            }

            string agentResponse = GetMockAgentResponse(request.Message);

            return Ok(new AgentMessageResponseDto
            {
                IsSuccess = true,
                Message = agentResponse
            });
        }

        private static string GetMockAgentResponse(string userMessage)
        {
            string normalizedMessage = userMessage
                .Trim()
                .ToLower(new System.Globalization.CultureInfo("tr-TR"));

            if (normalizedMessage.Contains("hava"))
            {
                return "Elbette, hava durumunu kontrol edebilirim. Hangi şehir için bilgi almak istiyorsunuz?";
            }

            if (normalizedMessage.Contains("otel"))
            {
                return "Size uygun otelleri listeleyebilirim. Şehir, tarih aralığı ve kişi sayısını paylaşır mısınız?";
            }

            if (normalizedMessage.Contains("restoran"))
            {
                return "Restoran önerileri hazırlayabilirim. Hangi şehirde ve hangi mutfak türünde restoran arıyorsunuz?";
            }

            if (normalizedMessage.Contains("rota"))
            {
                return "Memnuniyetle bir seyahat rotası oluşturabilirim. Şehir, gün sayısı ve seyahat tercihlerinizi paylaşabilirsiniz.";
            }

            if (normalizedMessage.Contains("döviz") ||
                normalizedMessage.Contains("kur") ||
                normalizedMessage.Contains("euro") ||
                normalizedMessage.Contains("dolar"))
            {
                return "Döviz hesaplaması yapabilirim. Çevirmek istediğiniz tutarı ve para birimlerini paylaşır mısınız?";
            }

            if (normalizedMessage.Contains("şehir"))
            {
                return "Size uygun bir şehir önerebilmem için seyahat tarihinizi, bütçenizi ve tatil tercihinizi öğrenmem gerekiyor.";
            }

            return "Talebinizi aldım. Gideceğiniz şehir, seyahat tarihi, kişi sayısı ve bütçe bilgilerini paylaşarak başlayabilirsiniz.";
        }

async function getAgentResponse(userMessage) {
    const response = await fetch("/Dashboard/SendMessage", {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify({
            message: userMessage
        })
    });

    let result;

    try {
        result = await response.json();
    } catch {
        throw new Error("Sunucudan geçerli bir cevap alınamadı.");
    }

    if (!response.ok || !result.isSuccess) {
        throw new Error(
            result.message || "Mesaj işlenirken bir hata oluştu."
        );
    }

    return result.message;
}

async function sendMessage() {
    const message = messageInput.value.trim();

    if (message === "") {
        messageInput.focus();
        return;
    }

    createUserMessage(message);

    messageInput.value = "";
    messageInput.focus();

    sendMessageButton.disabled = true;
    messageInput.disabled = true;

    showThinking();
    scrollChatToBottom();

    try {
        const agentResponse = await getAgentResponse(message);

        hideThinking();

        createAgentMessage(agentResponse);
    }
    catch (error) {
        hideThinking();

        createAgentMessage(
            error.message || "Üzgünüm, mesajınızı işlerken bir hata oluştu."
        );
    }
    finally {
        sendMessageButton.disabled = false;
        messageInput.disabled = false;

        messageInput.focus();
        scrollChatToBottom();
    }
}

                result = await response.json();
            } catch {
                throw new Error("Sunucudan geçerli bir cevap alınamadı.");
            }

            if (!response.ok) {
                throw new Error(
                    result.message || "Sunucu isteği başarısız oldu."
                );
            }

            if (!result.isSuccess) {
                throw new Error(
                    result.message || "Mesaj işlenirken bir hata oluştu."
                );
            }

            return result.message;
        }

        async function sendMessage() {
            const message = messageInput.value.trim();

            if (message === "") {
                messageInput.focus();
                return;
            }

            createUserMessage(message);

            messageInput.value = "";

            setLoadingState(true);
            showThinking();
            scrollChatToBottom();

            try {
                const agentResponse = await getAgentResponse(message);

                hideThinking();
                createAgentMessage(agentResponse);
            } catch (error) {
                hideThinking();

                createAgentMessage(
                    error.message ||
                    "Üzgünüm, mesajınızı işlerken bir hata oluştu."
                );
            } finally {
                setLoadingState(false);

                messageInput.focus();
                scrollChatToBottom();
            }
        }

        sendMessageButton.addEventListener("click", function () {
            sendMessage();
        });

        messageInput.addEventListener("keydown", function (event) {
            if (event.key === "Enter" && !event.shiftKey) {
                event.preventDefault();
                sendMessage();
            }
        });

        quickPromptButtons.forEach(function (button) {
            button.addEventListener("click", function () {
                if (messageInput.disabled) {
                    return;
                }

                messageInput.value = button.textContent.trim();
                messageInput.focus();
            });
        });

        scrollChatToBottom();
        messageInput.focus();
    });
</script>
