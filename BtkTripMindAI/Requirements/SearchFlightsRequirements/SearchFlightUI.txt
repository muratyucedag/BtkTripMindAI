function formatFlightTime(dateValue) {

    if (!dateValue)
        return "--:--";

    const date = new Date(dateValue);

    return new Intl.DateTimeFormat(
        "tr-TR",
        {
            hour: "2-digit",
            minute: "2-digit"
        }
    ).format(date);
}


function formatDuration(minutes) {

    const value = Number(minutes) || 0;

    if (value <= 0)
        return "-";

    const hours = Math.floor(value / 60);
    const remainingMinutes = value % 60;

    if (hours === 0)
        return `${remainingMinutes} dk`;

    return `${hours} sa ${remainingMinutes} dk`;
}


function createBaggageText(flight) {

    const carryOn =
        flight.carryOnBagCount;

    const checked =
        flight.checkedBagCount;

    if (carryOn > 0 && checked > 0) {
        return `${carryOn} kabin + ${checked} kayıtlı bagaj`;
    }

    if (carryOn > 0) {
        return `${carryOn} kabin bagajı`;
    }

    if (checked > 0) {
        return `${checked} kayıtlı bagaj`;
    }

    return "Bagaj bilgisi belirtilmemiş";
}


function getAirlineInitials(airline) {

    if (!airline)
        return "✈";

    return airline
        .split(" ")
        .filter(x => x)
        .slice(0, 2)
        .map(x => x[0])
        .join("")
        .toUpperCase();
}


function escapeHtml(value) {

    if (!value)
        return "";

    return String(value)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
}

private static string NormalizeAirportQuery(string query)
{
    string normalizedQuery = query.Trim();

    string[] removableWords =
    {
        " Uluslararası Havalimanı",
        " Uluslararası Havaalanı",
        " International Airport",
        " Havalimanı",
        " Havaalanı",
        " Airport"
    };

    foreach (string word in removableWords)
    {
        normalizedQuery = normalizedQuery.Replace(
            word,
            "",
            StringComparison.OrdinalIgnoreCase);
    }

    return normalizedQuery.Trim();
}
